1. Stop any Codex session started with `make codex` using its normal exit action.
   Do not kill unrelated sessions.
2. Open a normal terminal and run
   `cd /Users/pwiejak/Desktop/tools`.
3. Run the following block. It verifies the repository, archive integrity,
   contained paths and every current file hash before any write. It refuses
   restoration if a later edit or a symlink is present. Such a refusal requires
   reviewing those later edits; do not remove the checks.

```sh
/usr/bin/python3 -B - <<'PY'
import hashlib
import json
from pathlib import Path
import subprocess

root = Path('/Users/pwiejak/Desktop/tools')
actual = subprocess.check_output(
    ['git', 'rev-parse', '--show-toplevel'], cwd=root, text=True).strip()
if actual != str(root):
    raise SystemExit('Wrong repository; nothing changed')
backup = root / 'logs/change-backups/2026-10-28-codex'
for path in (root / 'logs', root / 'logs/change-backups', backup,
             backup / 'before.json', backup / 'after.json'):
    if path.is_symlink():
        raise SystemExit('Symlinked recovery path; nothing changed')
original_bytes = (backup / 'before.json').read_bytes()
state = json.loads((backup / 'after.json').read_text())
if hashlib.sha256(original_bytes).hexdigest() != state['before_sha256']:
    raise SystemExit('Recovery archive changed; nothing changed')
original = json.loads(original_bytes)['before']
if set(original) != set(state['files']):
    raise SystemExit('Recovery manifest paths differ; nothing changed')

for name, expected in state['files'].items():
    relative = Path(name)
    if relative.is_absolute() or '..' in relative.parts:
        raise SystemExit('Invalid recovery path; nothing changed')
    path = root / relative
    for parent in (path, *path.parents):
        if parent == root:
            break
        if parent.is_symlink():
            raise SystemExit('Symlink found; nothing changed: ' + name)
    if path.exists() and not path.is_file():
        raise SystemExit('Unexpected file type; nothing changed: ' + name)
    digest = hashlib.sha256(path.read_bytes()).hexdigest() if path.exists() else None
    if digest != expected:
        raise SystemExit('Later edit found; nothing changed: ' + name)

for name, content in original.items():
    path = root / name
    if content is None:
        if path.exists():
            path.unlink()
    else:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(content)
        path.chmod(0o755 if name == 'codex.sh' else 0o644)
print('Captured pre-change text restored. Recovery archive retained.')
PY
```
